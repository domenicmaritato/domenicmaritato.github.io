Command-line LZW compression/decompression tool with configurable code-width bounds and a table-reset policy for handling dictionary overflow.

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.*;

public class LZWTool {

	public static void main(String[] args) throws IOException {

		String alphabetPath = null;
		String policy = "freeze";
		String mode = null;
		int maxBits = 16;
		int minBits = 9;

		for (int i = 0; i < args.length; i++) {
			switch (args[i]) {
				case "--mode": {
					mode = args[++i];
					break;
				}
				case "--minW": {
					minBits = Integer.parseInt(args[++i]);
					break;
				}
				case "--maxW": {
					maxBits = Integer.parseInt(args[++i]);
					break;
				}
				case "--policy": {
					policy = args[++i];
					break;
				}
				case "--alphabet": {
					alphabetPath = args[++i];
					break;
				}
				default: {
					System.err.println("Unknown argument: " + args[i]);
					System.exit(2);
				}
			}
		}

		if ("expand".equals(mode)) {
			decodeStream();
		} else {
			List<String> syms = readSymbolFile(alphabetPath);
			encodeStream(syms, minBits, maxBits, policy);
		}
	}

	static List<String> readSymbolFile(String path) throws IOException {

		List<String> syms = new ArrayList<>();
		LinkedHashSet<String> dupeCheck = new LinkedHashSet<>();

		// Use BufferedReader instead of Scanner so that U+0085 (NEL) and other
		// Unicode line-separator chars are not consumed as line terminators.
		// BufferedReader.readLine() only splits on \n, \r, and \r\n.
		BufferedReader br = new BufferedReader(
				new InputStreamReader(new FileInputStream(path), StandardCharsets.UTF_8));
		String ln;
		while ((ln = br.readLine()) != null) {
			if (ln.isEmpty()) {
				continue;
			}
			String ch = String.valueOf(ln.charAt(0)); // take first char of each line
			if (dupeCheck.add(ch)) {
				syms.add(ch);
			}
		}
		br.close();

		if (dupeCheck.add(String.valueOf((char) 10))) {
			syms.add(String.valueOf((char) 10));
		}

		if (dupeCheck.add(String.valueOf((char) 13))) {
			syms.add(String.valueOf((char) 13));
		}

		return syms;
	}

	private static void encodeStream(List<String> syms, int minBits, int maxBits, String tablePolicy) {

		BinaryStdOut.write(minBits - 1, 4);
		BinaryStdOut.write(maxBits - 1, 4);

		if (tablePolicy.equals("reset")) {
			BinaryStdOut.write(1, 1);
		} else {
			BinaryStdOut.write(0, 1);
		}

		// Write symCount as 8 bits; 256 symbols wraps to 0 (& 0xFF), decoder treats 0 as 256.
		BinaryStdOut.write(syms.size() & 0xFF, 8);

		for (String s : syms) {
			BinaryStdOut.write((byte) s.charAt(0));
		}

		TSTmod<Integer> dict = new TSTmod<>();
		for (int i = 0; i < syms.size(); i++) {
			dict.put(new StringBuilder(syms.get(i)), i);
		}

		// syms.size() = EOF marker (last code written, tells decoder to stop)
		// syms.size()+1 = reset marker (written before wiping the table)
		// syms.size()+2 = first real dictionary entry
		int eofCode = syms.size();
		int resetCode = syms.size() + 1;
		int freeSlot = syms.size() + 2;
		int curWidth = minBits;
		int widthCap = 1 << curWidth;

		// bump width until both sentinel codes fit in curWidth bits
		while (widthCap <= resetCode && curWidth < maxBits) {
			curWidth++;
			widthCap = 1 << curWidth;
		}

		StringBuilder match = new StringBuilder();

		while (!BinaryStdIn.isEmpty()) {
			char incoming = (char) (BinaryStdIn.readByte() & 0xFF);
			StringBuilder longer = new StringBuilder(match).append(incoming);

			if (dict.contains(longer)) {
				match = longer;
				continue;
			}

			// if match is empty or the char isn't in the alphabet, skip it
			if (match.length() == 0 || dict.get(match) == null) {
				match = new StringBuilder().append(incoming);
				continue;
			}

			BinaryStdOut.write(dict.get(match), curWidth);

			if (freeSlot < (1 << maxBits)) {
				dict.put(longer, freeSlot++);

				if (freeSlot > widthCap && curWidth < maxBits) {
					curWidth++;
					widthCap = 1 << curWidth;
				}
			} else {
				if (tablePolicy.equals("reset")) {
					BinaryStdOut.write(resetCode, curWidth); // tell decoder to reset too

					dict = new TSTmod<>();
					for (int i = 0; i < syms.size(); i++) {
						dict.put(new StringBuilder(syms.get(i)), i);
					}

					freeSlot = syms.size() + 2;
					curWidth = minBits;
					widthCap = 1 << curWidth;

					// bump width again after reset so sentinels still fit
					while (widthCap <= resetCode && curWidth < maxBits) {
						curWidth++;
						widthCap = 1 << curWidth;
					}
				}
			}

			match = new StringBuilder().append(incoming);
		}

		if (match.length() > 0 && dict.get(match) != null) {
			BinaryStdOut.write(dict.get(match), curWidth);
		}

		BinaryStdOut.write(eofCode, curWidth); // marks end of stream for decoder

		BinaryStdOut.close();
	}

	private static void decodeStream() {

		int minBits = BinaryStdIn.readInt(4) + 1;
		int maxBits = BinaryStdIn.readInt(4) + 1;

		BinaryStdIn.readInt(1); // policy bit - not needed, resets come through as resetCode

		int symCount = BinaryStdIn.readInt(8);
		// 256 symbols wraps to 0 in 8 bits; treat 0 as 256
		if (symCount == 0) symCount = 256;

		List<String> syms = new ArrayList<>();

		for (int i = 0; i < symCount; i++) {
			syms.add(String.valueOf((char) (BinaryStdIn.readByte() & 0xFF)));
		}

		String[] lookupTable = new String[1 << maxBits];

		for (int i = 0; i < syms.size(); i++) {
			lookupTable[i] = syms.get(i);
		}

		int eofCode = syms.size();
		int resetCode = syms.size() + 1;
		int curWidth = minBits;
		int freeSlot = syms.size() + 2;
		int widthCap = 1 << curWidth;

		// bump width until both sentinel codes fit in curWidth bits
		while (widthCap <= resetCode && curWidth < maxBits) {
			curWidth++;
			widthCap = 1 << curWidth;
		}

		// read first code separately - no table entry gets added for it
		int codeword = BinaryStdIn.readInt(curWidth);
		if (codeword == eofCode) {
			BinaryStdOut.close();
			return;
		}
		String decoded = lookupTable[codeword];
		BinaryStdOut.write(decoded);

		while (!BinaryStdIn.isEmpty()) {

			String lastDecoded = decoded;
			codeword = BinaryStdIn.readInt(curWidth);

			if (codeword == eofCode) {
				break;
			}

			if (codeword == resetCode) {
				lookupTable = new String[1 << maxBits];
				for (int i = 0; i < syms.size(); i++) {
					lookupTable[i] = syms.get(i);
				}

				freeSlot = syms.size() + 2;
				curWidth = minBits;
				widthCap = 1 << curWidth;

				// bump width again after reset so sentinels still fit
				while (widthCap <= resetCode && curWidth < maxBits) {
					curWidth++;
					widthCap = 1 << curWidth;
				}

				// treat the first code after reset like the initial read above:
				// output it but don't add a table entry
				if (BinaryStdIn.isEmpty()) {
					break;
				}
				codeword = BinaryStdIn.readInt(curWidth);
				if (codeword == eofCode) {
					break;
				}
				decoded = lookupTable[codeword];
				BinaryStdOut.write(decoded);
				continue;
			}

			if (codeword == freeSlot) {
				decoded = lastDecoded + lastDecoded.charAt(0);
			} else {
				decoded = lookupTable[codeword];
			}

			BinaryStdOut.write(decoded);

			if (freeSlot < (1 << maxBits)) {
				lookupTable[freeSlot] = lastDecoded + decoded.charAt(0);
				freeSlot++;

				if (freeSlot == widthCap && curWidth < maxBits) {
					curWidth++;
					widthCap = 1 << curWidth;
				}
			}
		}

		BinaryStdOut.close();
	}
}

DLBHotspotDetector.java

Java · 249 lines

Builds a de la Briandais trie from leaked passwords to detect risky substrings ("hotspots") in candidate passwords, tracking frequency and position stats.

import java.util.Set;
import java.util.HashSet;
import java.util.HashMap;
import java.util.Map;

public class DLBHotspotDetector implements HotspotDetector {

    private Node root;

    private static class Node {
        char ch;
        Node child;
        Node sibling;
        boolean isTerminal;

        int freq; // total number of times password appeared
        int docFreq; // number of passwords containing string
        int beginCount; // number of times string is at beginning
        int middleCount;// number of times string is at middle
        int endCount;// number of times string is at end

        Node(char ch) {
            this.ch = ch;
            this.child = null;
            this.sibling = null;
            this.isTerminal = false;

            this.freq = 0;
            this.docFreq = 0;
            this.beginCount = 0;
            this.middleCount = 0;
            this.endCount = 0;
        }
    }

    public DLBHotspotDetector() {
        root = null;
    }

    // move through sibling list until matching character found
    private Node findSibling(Node node, char c) {

        // search horizontally across siblings for matching character
        while (node != null && node.ch != c) {
            node = node.sibling;
        }

        return node;
    }

    // move through sibling list until matching character found or created
    private Node findOrCreateSibling(Node node, char c) {

        //check
        if(node == null) {
            return new Node(c);
        }

        Node current = node;

        // move through sibling list until matching character found or created
        while (true) {

            if (current.ch == c)
                return current;

            if (current.sibling == null) {
                current.sibling = new Node(c);
                return current.sibling;
            }

            current = current.sibling;
        }
    }

    // insert a word into the DLB
    // returns the terminal node for that word
    private Node insert(String word) {

        // if tree empty, create root node using first character
        if (root == null) {
            root = new Node(word.charAt(0));
        }

        Node current = root;

        for (int index = 0; index < word.length(); index++) {

            char c = word.charAt(index);

            // move through sibling list until matching character found or created
            current = findOrCreateSibling(current, c);

            // if last character, mark terminal and return node
            if (index == word.length() - 1) {
                current.isTerminal = true;
                return current;
            }

            // move downward to child node for next character
            if (current.child == null) {
                current.child = new Node(word.charAt(index + 1));
            }

            current = current.child;
        }

        return null;
    }

    // search for a word in the DLB
    private Node searchNode(String word) {

        Node current = root;

        for (int index = 0; index < word.length(); index++) {

            char c = word.charAt(index);

            // search horizontally across siblings for matching character
            current = findSibling(current, c);

            if (current == null)
                return null;

            // if last character, only return if it forms a valid complete substring
            if (index == word.length() - 1) {
                return current.isTerminal ? current : null;
            }

            // move down one level to continue matching next character
            current = current.child;
        }

        return null;
    }

    @Override
    public void addLeakedPassword(String leakedPassword, int minN, int maxN) {

        if (minN <= 0 || maxN <= 0 || minN > maxN) {
            throw new IllegalArgumentException();
        }

        if (leakedPassword == null)
            return;


        int length = leakedPassword.length();

        Set<Node> seen = new HashSet<>();

        // generate all substrings within allowed size range
        for (int start = 0; start < length; start++) {

            for (int size = minN; size <= maxN; size++) {

                if (start + size > length)
                    break;

                String sub = leakedPassword.substring(start, start + size);

                Node terminal = insert(sub);

                // increment total frequency every time substring appears
                terminal.freq++;

                if (!seen.contains(terminal)) {
                    terminal.docFreq++;
                    seen.add(terminal);
                }

                // record where substring appears in leaked password
                if (start == 0) {
                    terminal.beginCount++;
                } else if (start + size == length) {
                    terminal.endCount++;
                } else {
                    terminal.middleCount++;
                }
            }
        }
    }

    @Override
    public Set<Hotspot> hotspotsIn(String candidatePassword) {

        Set<Hotspot> result = new HashSet<>();

        if (candidatePassword == null)
            return result;

        int length = candidatePassword.length();

        // use hashmap to prevent duplicates
        Map<String, Hotspot> map = new HashMap<>();

        // generate candidate substrings within allowed size range
        for (int start = 0; start < length; start++) {

            for (int size = 1; start + size <= length; size++) {

                if (start + size > length)
                    break;

                int end = start + size;

                String sub = candidatePassword.substring(start, end);

                // lookup substring in DLB to see if it exists in the data
                Node terminal = searchNode(sub);

                if (terminal == null)
                    continue;

                Hotspot h = map.get(sub);

                // if first time seeing substring, create new Hotspot object
                if (h == null) {
                    boolean atBegin = (start == 0);
                    boolean atEnd = (end == length);

                    int middle = (!atBegin && !atEnd) ? 1 : 0;

                    h = new Hotspot(sub, terminal.freq, terminal.docFreq, terminal.beginCount, terminal.middleCount,
                            terminal.endCount, atBegin, middle, atEnd);

                    // store hotspot in map.. prevents duplicates
                    map.put(sub, h);
                } else {

                    // update candidate position
                    if (start == 0)
                        h.candidateAtBegin = true;

                    else if (end == length)
                        h.candidateAtEnd = true;

                    else
                        h.candidateMiddleCount++;
                }
            }
        }

        // convert map values into final result set
        result.addAll(map.values());

        return result;
    }
}

PittGuard.java

Graph tool that models network vulnerability: BFS to trace infection paths between compromised nodes, and Dijkstra to find optimal patch-server routing.

import java.util.*;
import java.io.*;

public class PittGuard {

    public static void main(String[] args) throws Exception {

        // grab all the command line stuff
        String mode = null;
        String fileName = null;
        String startNode = null;
        String endNode = null;
        String serverNode = null;
        boolean directed = false;

        int i = 0;
        while (i < args.length) {
            if (args[i].equals("--mode")) {
                mode = args[i + 1];
                i += 2;
            } else if (args[i].equals("--input")) {
                fileName = args[i + 1];
                i += 2;
            } else if (args[i].equals("--src")) {
                startNode = args[i + 1];
                i += 2;
            } else if (args[i].equals("--dst")) {
                endNode = args[i + 1];
                i += 2;
            } else if (args[i].equals("--server")) {
                serverNode = args[i + 1];
                i += 2;
            } else if (args[i].equals("--directed")) {
                directed = true;
                i++;
            } else {
                i++;
            }
        }

        if (mode == null || fileName == null) {
            System.err.println("Error: missing --mode or --input");
            System.exit(1);
        }

        // open the file
        Scanner sc = new Scanner(new File(fileName));

        // skip blank lines and comments to get the node count
        int numNodes = 0;
        while (sc.hasNextLine()) {
            String line = sc.nextLine().trim();
            if (!line.isEmpty() && !line.startsWith("#")) {
                numNodes = Integer.parseInt(line);
                break;
            }
        }

        // hackable = true means the node is vulnerable
        HashMap<String, Boolean> hackable = new HashMap<>();
        int count = 0;
        while (sc.hasNextLine() && count != numNodes) {
            String line = sc.nextLine().trim();
            if (line.isEmpty() || line.startsWith("#"))
                continue;
            String[] words = line.split("\\s+");
            hackable.put(words[0], Boolean.parseBoolean(words[1]));
            count++;
        }

        // build the graph - each entry is {neighbor, latency, encryption}
        HashMap<String, ArrayList<String[]>> graph = new HashMap<>();
        for (String node : hackable.keySet()) {
            graph.put(node, new ArrayList<>());
        }

        // read all the edges
        while (sc.hasNextLine()) {
            String line = sc.nextLine().trim();
            if (line.isEmpty() || line.startsWith("#"))
                continue;
            String[] words = line.split("\\s+");
            if (words.length < 4)
                continue;
            String u = words[0];
            String v = words[1];
            String lat = words[2];
            String enc = words[3];
            if (directed) {
                graph.get(u).add(new String[] { v, lat, enc });
            } else {
                graph.get(u).add(new String[] { v, lat, enc });
                graph.get(v).add(new String[] { u, lat, enc });
            }
        }
        sc.close();

        if (mode.equals("infect")) {
            runBFS(startNode, endNode, hackable, graph);
        } else if (mode.equals("patch")) {
            runDijkstra(serverNode, hackable, graph);
        } else {
            System.err.println("Error: unknown mode " + mode);
            System.exit(1);
        }
    }

    // task 1 - bfs but only through vulnerable nodes
    static void runBFS(String startNode, String endNode, HashMap<String, Boolean> hackable, HashMap<String, ArrayList<String[]>> graph) {

        if (startNode == null || endNode == null) {
            System.err.println("Error: need --src and --dst");
            System.exit(1);
        }
        if (!hackable.containsKey(startNode) || !hackable.containsKey(endNode)) {
            System.err.println("Error: unknown node");
            System.exit(1);
        }

        // if either endpoint isnt vulnerable just return -1
        if (!hackable.get(startNode) || !hackable.get(endNode)) {
            System.out.println(-1);
            return;
        }

        if (startNode.equals(endNode)) {
            System.out.println(0);
            return;
        }

        // jumpCount tracks how many hops to reach each node
        HashMap<String, Integer> jumpCount = new HashMap<>();
        Queue<String> toVisit = new LinkedList<>();
        jumpCount.put(startNode, 0);
        toVisit.add(startNode);

        while (!toVisit.isEmpty()) {
            String current = toVisit.poll();
            int stepsSoFar = jumpCount.get(current);

            for (String[] edge : graph.get(current)) { //relax
                String neighbor = edge[0];
                // skip non-vulnerable neighbors
                if (hackable.get(neighbor) && jumpCount.get(neighbor) == null) {
                    jumpCount.put(neighbor, stepsSoFar + 1);
                    toVisit.add(neighbor);
                }
            }
        }

        // check if we made it to endNode
        if (jumpCount.containsKey(endNode)) {
            System.out.println(jumpCount.get(endNode));
        } else {
            System.out.println(-1);
        }
    }

    // task 2 - dijkstra from patch server to all vulnerable nodes
    static void runDijkstra(String serverNode, HashMap<String, Boolean> hackable, HashMap<String, ArrayList<String[]>> graph) {

        if (serverNode == null) {
            System.err.println("Error: need --server");
            System.exit(1);
        }
        if (!hackable.containsKey(serverNode)) {
            System.err.println("Error: server node doesnt exist");
            System.exit(1);
        }
        if (hackable.get(serverNode)) {
            System.err.println("Error: server cant be vulnerable");
            System.exit(1);
        }

        // distances starts everything at infinity
        HashMap<String, Double> distances = new HashMap<>();
        for (String node : hackable.keySet()) {
            distances.put(node, Double.MAX_VALUE);
        }
        distances.put(serverNode, 0.0);

        // priority queue that picks the cheapest node next
        Comparator<String> byDist = new Comparator<String>() {
            public int compare(String a, String b) {
                return Double.compare(distances.get(a), distances.get(b));
            }
        };

        PriorityQueue<String> pq = new PriorityQueue<>(byDist);
        pq.add(serverNode);

        HashSet<String> seen = new HashSet<>();

        while (!pq.isEmpty()) {
            String current = pq.poll();

            if (seen.contains(current))
                continue;
            seen.add(current);

            double myCost = distances.get(current);

            for (String[] edge : graph.get(current)) {
                String neighbor = edge[0];
                double latency = Double.parseDouble(edge[1]);
                int enc = Integer.parseInt(edge[2]);

                // effective cost formula from the spec
                double realCost = latency * (1.0 + (3 - enc) / 10.0);
                double newCost = myCost + realCost;

                if (newCost < distances.get(neighbor)) {
                    distances.put(neighbor, newCost);
                    pq.add(neighbor);
                }
            }
        }

        // first check if any vulnerable node is totally unreachable
        for (String node : hackable.keySet()) {
            if (hackable.get(node) && distances.get(node) == Double.MAX_VALUE) {
                System.out.println("INF");
                return;
            }
        }

        // then grab the biggest distance among vulnerable nodes
        double biggestDist = 0.0;
        for (String node : hackable.keySet()) {
            if (hackable.get(node)) {
                if (distances.get(node) > biggestDist)
                    biggestDist = distances.get(node);
            }
        }

        System.out.printf("%.1f%n", biggestDist);
    }
}

PittNetShield.java

Java · 105 lines

Uses Kruskal's algorithm with union-find to compute the minimum-risk spanning structure of a network from edge failure probabilities.

import java.util.*;
import java.io.*;

public class PittNetShield {

    static int[] parent;
    static int[] rank;

    static int find(int n) {
        //recursively find the root of the set and do path compression
        if (parent[n] != n) {
            parent[n] = find(parent[n]);
        }

        return parent[n];
    }

    static void union(int x, int y) {
        int rootX = find(x), rootY = find(y);
        if (rootX == rootY)
            return;
        if (rank[rootX] < rank[rootY])
            parent[rootX] = rootY;
        else if (rank[rootX] > rank[rootY])
            parent[rootY] = rootX;
        else {
            parent[rootY] = rootX;
            rank[rootX]++;
        }
    }

    public static void main(String[] args) throws Exception {
        Scanner scanner = new Scanner(new File(args[0]));

        int numNodes = scanner.nextInt();
        scanner.nextLine();

        Map<String, Integer> nodeMap = new HashMap<>();

        for (int i = 0; i < numNodes; i++) {
            nodeMap.put(scanner.nextLine().trim(), i);
        }

        List<double[]> edgeList = new ArrayList<>();

        while (scanner.hasNextLine()) {

            String line = scanner.nextLine().trim();
            if (line.isEmpty())
                continue;
            String[] tokens = line.split("\\s+");
            int src = nodeMap.get(tokens[0]);
            int dst = nodeMap.get(tokens[1]);

            double prob = Double.parseDouble(tokens[2]);
            // convert probability to a weight
            double weight = -Math.log(1 - prob);
            edgeList.add(new double[] {weight, src, dst});

        }

        if (numNodes == 1) {
            System.out.println("0");
            return;
        }

        Collections.sort(edgeList, new Comparator<double[]>() {
            // sort by weight... kruskal's
            public int compare(double[] a, double[] b) {
                return Double.compare(a[0], b[0]);
            }
        });

        parent = new int[numNodes];
        rank = new int[numNodes];

        for (int i = 0; i < numNodes; i++) {
            parent[i] = i;
        }

        double totalWeight = 0;
        int count = 0;

        for (double[] edge : edgeList) {
            // add the smallest edges that don't form a cycle
            int u = (int) edge[1], v = (int) edge[2];
            if (find(u) != find(v)) {
                union(u, v);
                totalWeight += edge[0];
                count++;
            }
            if (count == numNodes - 1)
                break;
        }

        if (count < numNodes - 1) {
            System.out.println("DISCONNECTED");
            return;
        }

        double notCaught = Math.exp(-totalWeight);
        double caught = 1 - notCaught;

        System.out.printf("%.2f%%%n", caught * 100);
    }
}

from sqlalchemy.orm import sessionmaker
from models.order_model import Base, Order

from flask import Flask, render_template, request, redirect, url_for, abort

DATABASE_URL = "sqlite:///order.db"

engine = create_engine(DATABASE_URL)
Base.metadata.create_all(engine)

Session = sessionmaker(bind=engine)
session = Session()


app = Flask(__name__)

@app.route("/")
def home():
    return redirect(url_for("main_menu", table_number=1))

@app.route("/<table_number>")
def main_menu(table_number):
    return render_template("main_menu.html", table_number=table_number)

@app.route("/order/<table_number>", methods=["POST"])
def order(table_number):
    customer_name = request.form.get("customer_name")
    order_items = request.form.get("order_summary")
    if order_items:
        order_items = order_items.splitlines()
    else:
        order_items = []

    orders_string = ",".join(order_items)

    new_order = Order(
        customer_name=customer_name,
        orders=orders_string,
        table_number=table_number
    )

    try:
        session.add(new_order)
        session.commit()
        print(f"Order {new_order.id} saved successfully!")
    except Exception as e:
        session.rollback()
        print(f"error: {e}")

    data = {
        "customer_name": customer_name,
        "order_items": order_items,
        "table_number": table_number
    }

    return render_template("order_summary.html", data=data)

@app.route("/kitchen/")
def kitchen():
    orders = session.query(Order).order_by(Order.date_created).all()
    orders_json = []
    for order in orders:
        orders_json.append(order.toJSON())
    print(orders_json)
    return render_template("kitchen.html", orders=orders_json)

@app.route('/delete_order/<order_id>', methods=['POST'])
def delete_order(order_id):
    order_to_delete = session.query(Order).get(order_id)
    if order_to_delete is None:
        abort(404)
    try:
        session.delete(order_to_delete)
        session.commit()
        return redirect(url_for("kitchen"))
    except:
        print("error")
        return "error"

if __name__ == "__main__":
    app.run(debug=True)
